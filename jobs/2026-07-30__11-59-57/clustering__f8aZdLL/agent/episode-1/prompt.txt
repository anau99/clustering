def hash_encoder(df: pd.DataFrame, columns=None, n_features=8) -> pd.DataFrame:
    """
    Hash categorical columns into fixed-size integer features.
    """

    result = df.copy()

    if columns is None:
        columns = result.select_dtypes(include=["object", "category"]).columns.tolist()
