> def select_kmeans(X, seeds=(0, 1, 2, 3, 4, 7, 11, 42, 101, 202), min_prop=0.1):
>     """Pick KMeans run by silhouette on PCA space with a penalty for tiny clusters."""
>     # PCA for speed/stability in silhouette computation
>     n_comp = int(min(10, X.shape[1]))
>     Z = PCA(n_components=n_comp, random_state=0).fit_transform(X)
>
>     best = None
>     best_score = -1e9
>
>     for seed in seeds:
>         km = KMeans(n_clusters=2, n_init=50, random_state=seed)
>         lab = km.fit_predict(X)
>
>         # balance penalty
>         counts = np.bincount(lab, minlength=2)
>         props = counts / counts.sum()
>         small = props.min()
>         penalty = 0.0
>         if small < min_prop:
>             penalty = (min_prop - small) * 5.0  # strong penalty
>
>         # silhouette on reduced space
>         if len(np.unique(lab)) < 2:
>             continue
>         sil = silhouette_score(Z, lab)
>         score = sil - penalty
>
>         if score > best_score:
>             best_score = score
>             best = lab
>
>     return best
>
