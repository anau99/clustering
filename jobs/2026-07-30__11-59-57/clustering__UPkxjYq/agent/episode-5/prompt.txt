import numpy as np
import pandas as pd

from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.cluster import KMeans


def kmeans_best_inertia(X, seeds=(0, 1, 2, 3, 4, 7, 11, 42)):
    best_lab = None
    best_inertia = np.inf
    for seed in seeds:
        km = KMeans(n_clusters=2, n_init=50, random_state=seed)
        lab = km.fit_predict(X)
        if km.inertia_ < best_inertia:
            best_inertia = km.inertia_
            best_lab = lab
    return best_lab


def cluster_diabetes(in_path, out_path):
    df = pd.read_csv(in_path)
    cat_cols = [c for c in df.columns if c != "age"]
    num_cols = ["age"]

    pre = ColumnTransformer(
        transformers=[
            ("num", StandardScaler(), num_cols),
            ("cat", OneHotEncoder(handle_unknown="ignore", sparse_output=False), cat_cols),
        ],
        remainder="drop",
    )

    X = pre.fit_transform(df)
    labels = kmeans_best_inertia(X)
    pd.DataFrame({"pseudo_label": labels.astype(int)}).to_csv(out_path, index=False)


def cluster_heart(in_path, out_path):
    df = pd.read_csv(in_path)

    cat_cols = ["sex", "cp", "fbs", "restecg", "exang", "slope", "ca", "thal"]
    num_cols = ["age", "trestbps", "chol", "thalach", "oldpeak"]

    pre = ColumnTransformer(
        transformers=[
            ("num", StandardScaler(), num_cols),
            ("cat", OneHotEncoder(handle_unknown="ignore", sparse_output=False), cat_cols),
        ],
        remainder="drop",
    )

    X = pre.fit_transform(df)
    labels = kmeans_best_inertia(X)
    pd.DataFrame({"pseudo_label": labels.astype(int)}).to_csv(out_path, index=False)


if __name__ == "__main__":
